/**
 * Registro Fácil by DayIA — "ponte" entre o app e o seu Google Drive
 *
 * O que este código faz:
 *  - Guarda cada registro (romaneio, nota fiscal, coleta…) numa pasta própria no seu Drive,
 *    dentro de uma pasta por mês, com o arquivo "Detalhes do registro.txt".
 *  - Anota tudo numa planilha criada sozinha ("Registro Fácil - Registros").
 *  - A data e a hora são SEMPRE as do servidor do Google (não as do celular).
 *  - Não existe apagar nem editar: nada some.
 *
 * Como usar (resumo):
 *  1) Troque a SENHA logo abaixo.
 *  2) Escolha a função "preparar" lá em cima e clique em Executar (autorize com a sua conta).
 *  3) Implantar → Nova implantação → App da Web → Executar como: Eu → Quem pode acessar: Qualquer pessoa.
 *  4) Copie o link que termina em /exec e cole no index.html (no lugar de COLE_AQUI_O_LINK_DO_EXEC).
 */

const SENHA = 'troque-esta-senha';          // ← troque por uma senha só sua
const FUSO = 'America/Sao_Paulo';
const NOME_PASTA = 'Registro Fácil by DayIA';
const NOME_PLANILHA = 'Registro Fácil - Registros';
const TIPOS = ['Romaneio', 'Nota fiscal', 'Coleta agendada', 'Outro'];
const MESES = ['janeiro', 'fevereiro', 'março', 'abril', 'maio', 'junho', 'julho', 'agosto', 'setembro', 'outubro', 'novembro', 'dezembro'];
const CAB_REG = ['ID', 'Data', 'Hora', 'Tipo', 'Cliente', 'Nº nota fiscal', 'Nº coleta', 'Observação', 'Arquivos', 'Situação', 'Link da pasta', 'ID da pasta', 'Última atualização'];
const CAB_ARQ = ['ID do registro', 'Data', 'Hora', 'Nome do arquivo', 'Tamanho', 'Link', 'ID do arquivo'];

/* ============================== PREPARAR ============================== */
function preparar() {
  const P = PropertiesService.getScriptProperties();
  let pasta = null, planilha = null;
  try { pasta = DriveApp.getFolderById(P.getProperty('PASTA_ID')); } catch (e) {}
  if (!pasta) {
    const achou = DriveApp.getFoldersByName(NOME_PASTA);
    pasta = achou.hasNext() ? achou.next() : DriveApp.createFolder(NOME_PASTA);
    P.setProperty('PASTA_ID', pasta.getId());
  }
  try { planilha = SpreadsheetApp.openById(P.getProperty('PLANILHA_ID')); } catch (e) {}
  if (!planilha) {
    planilha = SpreadsheetApp.create(NOME_PLANILHA);
    DriveApp.getFileById(planilha.getId()).moveTo(pasta);
    P.setProperty('PLANILHA_ID', planilha.getId());
  }
  aba_(planilha, 'Registros', CAB_REG);
  aba_(planilha, 'Arquivos', CAB_ARQ);
  const sobra = planilha.getSheetByName('Página1') || planilha.getSheetByName('Sheet1');
  if (sobra && planilha.getSheets().length > 2) planilha.deleteSheet(sobra);
  if (!P.getProperty('ULTIMO_NUMERO')) P.setProperty('ULTIMO_NUMERO', '0');
  console.log('✅ Tudo pronto!');
  console.log('📁 Pasta: ' + pasta.getUrl());
  console.log('📊 Planilha: ' + planilha.getUrl());
  if (SENHA === 'troque-esta-senha') console.log('⚠️ Lembre de trocar a SENHA lá no começo do código e salvar.');
  return { pasta: pasta, planilha: planilha };
}
function aba_(ss, nome, cab) {
  let sh = ss.getSheetByName(nome);
  if (!sh) sh = ss.insertSheet(nome);
  if (sh.getLastRow() === 0) {
    sh.appendRow(cab);
    sh.getRange(1, 1, 1, cab.length).setFontWeight('bold').setBackground('#0E2238').setFontColor('#FFFFFF');
    sh.setFrozenRows(1);
  }
  return sh;
}
function base_() {
  const P = PropertiesService.getScriptProperties();
  try {
    return { pasta: DriveApp.getFolderById(P.getProperty('PASTA_ID')), planilha: SpreadsheetApp.openById(P.getProperty('PLANILHA_ID')) };
  } catch (e) { return preparar(); }
}

/* ============================== ENTRADA ============================== */
function doGet() {
  return json_({ ok: true, app: 'Registro Fácil by DayIA', mensagem: 'A ponte está funcionando.' });
}
function doPost(e) {
  let p = {};
  try { p = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (x) { return json_({ ok: false, erro: 'pedido', mensagem: 'Pedido inválido.' }); }
  if (!senhaOk_(p.senha)) return json_({ ok: false, erro: 'senha', mensagem: 'Senha errada.' });
  try {
    switch (p.acao) {
      case 'entrar': return json_({ ok: true, agora: agora_() });
      case 'criar': return json_(criar_(p));
      case 'iniciar_envio': return json_(iniciarEnvio_(p));
      case 'parte': return json_(parte_(p));
      case 'situacao_envio': return json_(situacaoEnvio_(p));
      case 'concluir': return json_(concluir_(p));
      case 'listar': return json_(listar_());
      case 'compartilhar': return json_(compartilhar_(p));
      default: return json_({ ok: false, erro: 'acao', mensagem: 'Ação desconhecida.' });
    }
  } catch (err) {
    console.log('Erro em ' + p.acao + ': ' + err + (err.stack ? '\n' + err.stack : ''));
    return json_({ ok: false, erro: 'servidor', mensagem: String(err.message || err) });
  }
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function senhaOk_(s) {
  const c = CacheService.getScriptCache(), k = 'erros_senha';
  const erros = Number(c.get(k) || 0);
  if (erros >= 10) { Utilities.sleep(1500); return false; }   // muitas tentativas erradas: segura por 10 minutos
  if (String(s || '') === SENHA) return true;
  c.put(k, String(erros + 1), 600);
  return false;
}
function agora_() {
  const d = new Date();
  return { data: Utilities.formatDate(d, FUSO, 'dd/MM/yyyy'), hora: Utilities.formatDate(d, FUSO, 'HH:mm:ss'), iso: Utilities.formatDate(d, FUSO, "yyyy-MM-dd'T'HH:mm:ss") };
}
function limpa_(s, max) { return String(s == null ? '' : s).replace(/[\u0000-\u001F]/g, ' ').trim().slice(0, max || 200); }
function semPerigo_(v) { v = String(v); return /^[=+\-@]/.test(v) ? "'" + v : v; }   // evita fórmula na planilha
function nomeArquivo_(s) { return limpa_(s, 120).replace(/[\\/:*?"<>|#%]/g, '-'); }

/* ============================== CRIAR REGISTRO ============================== */
function criar_(p) {
  const tipo = TIPOS.indexOf(p.tipo) >= 0 ? p.tipo : 'Outro';
  const cliente = limpa_(p.cliente, 120);
  if (!cliente) return { ok: false, erro: 'cliente', mensagem: 'Preencha o cliente.' };
  const nf = limpa_(p.nf, 40), coleta = limpa_(p.coleta, 40), obs = limpa_(p.obs, 1500);
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const b = base_(), P = PropertiesService.getScriptProperties();
    // evita registro duplicado se o celular reenviar o mesmo pedido
    if (p.chave) {
      const ja = P.getProperty('CHAVE_' + p.chave);
      if (ja) { const r = acharRegistro_(ja); if (r) return Object.assign({ ok: true, repetido: true }, resumo_(r)); }
    }
    const n = Number(P.getProperty('ULTIMO_NUMERO') || 0) + 1;
    P.setProperty('ULTIMO_NUMERO', String(n));
    const id = 'R' + ('0000' + n).slice(-5);
    const d = new Date(), quando = agora_();
    const mes = Utilities.formatDate(d, FUSO, 'yyyy-MM') + ' (' + MESES[Number(Utilities.formatDate(d, FUSO, 'M')) - 1] + ')';
    const it = b.pasta.getFoldersByName(mes);
    const pastaMes = it.hasNext() ? it.next() : b.pasta.createFolder(mes);
    const pasta = pastaMes.createFolder(nomeArquivo_(id + ' - ' + tipo + ' - ' + cliente + (nf ? ' - NF ' + nf : '')));
    const sh = b.planilha.getSheetByName('Registros') || aba_(b.planilha, 'Registros', CAB_REG);
    sh.appendRow([id, quando.data, quando.hora, tipo, semPerigo_(cliente), semPerigo_(nf), semPerigo_(coleta), semPerigo_(obs), 0, 'Enviando arquivos…', pasta.getUrl(), pasta.getId(), quando.data + ' ' + quando.hora]);
    if (p.chave) P.setProperty('CHAVE_' + p.chave, id);
    const reg = { id: id, data: quando.data, hora: quando.hora, tipo: tipo, cliente: cliente, nf: nf, coleta: coleta, obs: obs, arquivos: 0, situacao: 'Enviando arquivos…', pastaUrl: pasta.getUrl(), pastaId: pasta.getId(), linha: sh.getLastRow() };
    escreverDetalhes_(reg, pasta, []);
    return Object.assign({ ok: true }, resumo_(reg));
  } finally { lock.releaseLock(); }
}
function resumo_(r) {
  return { id: r.id, data: r.data, hora: r.hora, tipo: r.tipo, cliente: r.cliente, nf: r.nf, coleta: r.coleta, obs: r.obs, arquivos: r.arquivos, situacao: r.situacao, pastaUrl: r.pastaUrl };
}
function acharRegistro_(id) {
  id = String(id || '').trim().toUpperCase();
  if (!/^R\d{5,}$/.test(id)) return null;
  const sh = base_().planilha.getSheetByName('Registros');
  const c = sh.getRange('A:A').createTextFinder(id).matchEntireCell(true).findNext();
  if (!c) return null;
  const v = sh.getRange(c.getRow(), 1, 1, CAB_REG.length).getValues()[0];
  const fd = x => x instanceof Date ? Utilities.formatDate(x, FUSO, 'dd/MM/yyyy') : String(x);
  const fh = x => x instanceof Date ? Utilities.formatDate(x, FUSO, 'HH:mm:ss') : String(x);
  const tira = x => String(x).replace(/^'/, '');
  return { id: v[0], data: fd(v[1]), hora: fh(v[2]), tipo: v[3], cliente: tira(v[4]), nf: tira(v[5]), coleta: tira(v[6]), obs: tira(v[7]), arquivos: Number(v[8]) || 0, situacao: v[9], pastaUrl: v[10], pastaId: v[11], linha: c.getRow() };
}

/* ============================== ENVIO DE ARQUIVOS (em partes de 5 MB) ============================== */
// Cria uma "sessão de envio" do Google Drive para um arquivo. O celular manda o arquivo em partes.
function iniciarEnvio_(p) {
  const reg = acharRegistro_(p.id);
  if (!reg) return { ok: false, erro: 'registro', mensagem: 'Registro não encontrado.' };
  const tamanho = Number(p.tamanho) || 0;
  if (tamanho <= 0) return { ok: false, erro: 'vazio', mensagem: 'Arquivo vazio.' };
  const tipo = limpa_(p.tipo, 100) || 'application/octet-stream';
  const meta = { name: nomeArquivo_(p.nome || 'arquivo'), parents: [reg.pastaId], mimeType: tipo };
  const cab = { Authorization: 'Bearer ' + ScriptApp.getOAuthToken(), 'X-Upload-Content-Type': tipo, 'X-Upload-Content-Length': String(tamanho) };
  if (/^https:\/\/[a-z0-9.\-]+(:\d+)?$/i.test(String(p.origem || ''))) cab.Origin = p.origem;   // libera o envio direto pelo celular
  const r = UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&supportsAllDrives=true&fields=id,name,size,webViewLink', {
    method: 'post', contentType: 'application/json; charset=UTF-8', payload: JSON.stringify(meta), headers: cab, muteHttpExceptions: true
  });
  if (r.getResponseCode() !== 200) return { ok: false, erro: 'drive', mensagem: 'O Drive não aceitou começar o envio (' + r.getResponseCode() + ').' };
  const h = r.getHeaders(), sessao = h.Location || h.location;
  if (!sessao) return { ok: false, erro: 'drive', mensagem: 'O Drive não devolveu o endereço de envio.' };
  return { ok: true, sessao: sessao };
}
function sessaoValida_(s) { return /^https:\/\/www\.googleapis\.com\/upload\/drive\/v3\/files\?/.test(String(s || '')); }
// Plano B: quando o celular não consegue mandar direto, a parte passa por aqui.
function parte_(p) {
  if (!sessaoValida_(p.sessao)) return { ok: false, erro: 'sessao', mensagem: 'Envio inválido.' };
  const bytes = Utilities.base64Decode(String(p.dados || ''));
  const inicio = Number(p.inicio) || 0, total = Number(p.total) || 0, fim = inicio + bytes.length - 1;
  const r = UrlFetchApp.fetch(p.sessao, {
    method: 'put', contentType: 'application/octet-stream', payload: bytes, muteHttpExceptions: true,
    headers: { 'Content-Range': 'bytes ' + inicio + '-' + fim + '/' + total }
  });
  return respostaEnvio_(r);
}
// Pergunta ao Drive quanto do arquivo já chegou (para continuar de onde parou).
function situacaoEnvio_(p) {
  if (!sessaoValida_(p.sessao)) return { ok: false, erro: 'sessao', mensagem: 'Envio inválido.' };
  const r = UrlFetchApp.fetch(p.sessao, { method: 'put', payload: '', muteHttpExceptions: true, headers: { 'Content-Range': 'bytes */' + (Number(p.total) || 0) } });
  return respostaEnvio_(r);
}
function respostaEnvio_(r) {
  const c = r.getResponseCode();
  if (c === 200 || c === 201) { let a = {}; try { a = JSON.parse(r.getContentText()); } catch (e) {} return { ok: true, fim: true, arquivo: a }; }
  if (c === 308) {
    const h = r.getHeaders(), range = h.Range || h.range || '';
    const m = String(range).match(/bytes=0-(\d+)/);
    return { ok: true, fim: false, proximo: m ? Number(m[1]) + 1 : 0 };
  }
  if (c === 404 || c === 410) return { ok: false, erro: 'expirou', mensagem: 'O envio expirou. Vou começar este arquivo de novo.' };
  return { ok: false, erro: 'drive', mensagem: 'O Drive respondeu ' + c + '.' };
}

/* ============================== CONCLUIR (atualiza detalhes e planilha) ============================== */
function concluir_(p) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const reg = acharRegistro_(p.id);
    if (!reg) return { ok: false, erro: 'registro', mensagem: 'Registro não encontrado.' };
    const b = base_(), pasta = DriveApp.getFolderById(reg.pastaId);
    const lista = arquivosDaPasta_(pasta);
    const falhas = (Array.isArray(p.falhas) ? p.falhas : []).map(function (f) { return limpa_(f, 120); }).filter(String).slice(0, 30);
    // anota na aba Arquivos os que ainda não estão lá
    const sa = b.planilha.getSheetByName('Arquivos') || aba_(b.planilha, 'Arquivos', CAB_ARQ);
    const jaTem = {};
    if (sa.getLastRow() > 1) sa.getRange(2, 7, sa.getLastRow() - 1, 1).getValues().forEach(function (v) { jaTem[v[0]] = 1; });
    lista.forEach(function (a) {
      if (!jaTem[a.id]) sa.appendRow([reg.id, a.data, a.hora, semPerigo_(a.nome), a.tamanhoTxt, a.url, a.id]);
    });
    const q = agora_();
    reg.arquivos = lista.length;
    reg.situacao = falhas.length ? 'Faltou: ' + falhas.join(', ') : (lista.length ? 'Completo ✅' : 'Sem arquivos');
    const sh = b.planilha.getSheetByName('Registros');
    sh.getRange(reg.linha, 9, 1, 2).setValues([[reg.arquivos, reg.situacao]]);
    sh.getRange(reg.linha, 13).setValue(q.data + ' ' + q.hora);
    escreverDetalhes_(reg, pasta, lista, falhas);
    return Object.assign({ ok: true }, resumo_(reg));
  } finally { lock.releaseLock(); }
}
function arquivosDaPasta_(pasta) {
  const out = [], it = pasta.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    if (f.getName() === 'Detalhes do registro.txt') continue;
    const d = f.getDateCreated(), t = f.getSize();
    out.push({ id: f.getId(), nome: f.getName(), url: f.getUrl(), tamanho: t, tamanhoTxt: tamanho_(t), data: Utilities.formatDate(d, FUSO, 'dd/MM/yyyy'), hora: Utilities.formatDate(d, FUSO, 'HH:mm:ss'), quando: d.getTime() });
  }
  return out.sort(function (a, b) { return a.quando - b.quando || (a.nome < b.nome ? -1 : 1); });
}
function tamanho_(b) { return b >= 1048576 ? (b / 1048576).toFixed(1).replace('.', ',') + ' MB' : Math.max(1, Math.round(b / 1024)) + ' KB'; }
function escreverDetalhes_(reg, pasta, lista, falhas) {
  const q = agora_();
  const linhas = [
    'REGISTRO FÁCIL by DayIA — comprovante',
    '======================================',
    'ID: ' + reg.id,
    'Feito em: ' + reg.data + ' às ' + reg.hora + ' (horário do servidor do Google, Brasília)',
    'Tipo: ' + reg.tipo,
    'Cliente: ' + reg.cliente,
    'Nº da nota fiscal: ' + (reg.nf || '-'),
    'Nº da coleta: ' + (reg.coleta || '-'),
    'Observação: ' + (reg.obs || '-'),
    '',
    'ARQUIVOS (' + lista.length + ') — data e hora em que cada um chegou no Drive:'
  ];
  if (!lista.length) linhas.push('  (nenhum arquivo ainda)');
  lista.forEach(function (a, i) { linhas.push('  ' + (i + 1) + '. ' + a.nome + ' — ' + a.tamanhoTxt + ' — recebido em ' + a.data + ' às ' + a.hora); });
  if (falhas && falhas.length) { linhas.push('', 'NÃO CHEGARAM (o celular avisou falha):'); falhas.forEach(function (f) { linhas.push('  - ' + f); }); }
  linhas.push('', 'Situação: ' + (reg.situacao || '-'), 'Última atualização: ' + q.data + ' às ' + q.hora, '', 'Este registro não pode ser apagado nem editado pelo app.');
  const texto = linhas.join('\n');
  const it = pasta.getFilesByName('Detalhes do registro.txt');
  if (it.hasNext()) it.next().setContent(texto); else pasta.createFile('Detalhes do registro.txt', texto, MimeType.PLAIN_TEXT);
}

/* ============================== HISTÓRICO E COMPARTILHAR ============================== */
function listar_() {
  const sh = base_().planilha.getSheetByName('Registros');
  const n = sh.getLastRow() - 1;
  if (n < 1) return { ok: true, registros: [] };
  const ini = Math.max(2, sh.getLastRow() - 1999);   // os 2.000 mais recentes
  const fd = x => x instanceof Date ? Utilities.formatDate(x, FUSO, 'dd/MM/yyyy') : String(x);
  const fh = x => x instanceof Date ? Utilities.formatDate(x, FUSO, 'HH:mm:ss') : String(x);
  const tira = x => String(x).replace(/^'/, '');
  const regs = sh.getRange(ini, 1, sh.getLastRow() - ini + 1, CAB_REG.length).getValues().map(function (v) {
    return { id: v[0], data: fd(v[1]), hora: fh(v[2]), tipo: v[3], cliente: tira(v[4]), nf: tira(v[5]), coleta: tira(v[6]), obs: tira(v[7]), arquivos: Number(v[8]) || 0, situacao: v[9], pastaUrl: v[10] };
  }).filter(function (r) { return r.id; }).reverse();
  return { ok: true, registros: regs };
}
function compartilhar_(p) {
  const reg = acharRegistro_(p.id);
  if (!reg) return { ok: false, erro: 'registro', mensagem: 'Registro não encontrado.' };
  const pasta = DriveApp.getFolderById(reg.pastaId);
  pasta.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);   // só para ver, ninguém edita
  return Object.assign({ ok: true, link: pasta.getUrl() }, resumo_(reg));
}
